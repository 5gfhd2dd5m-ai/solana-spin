    <script>
        const canvas = document.getElementById('wheel');
        const ctx = canvas.getContext('2d');
        const spinBtn = document.getElementById('spinBtn');
        const modal = document.getElementById('modal');
        const winAmountText = document.getElementById('winAmount');
        const connectBtn = document.getElementById('connectBtn');

        // Секторы колеса
        const sectors = [
            { label: '0.1 SOL', color: '#10b981' },
            { label: '0.5 SOL', color: '#e5a93c' },
            { label: '1 SOL', color: '#059669' },
            { label: '2 SOL', color: '#d97706' },
            { label: '3 SOL', color: '#10b981' },
            { label: '5 SOL', color: '#e5a93c' },
            { label: '10 SOL', color: '#059669' },
            { label: '50 SOL', color: '#d97706' }
        ];

        const numSectors = sectors.length;
        const arc = (2 * Math.PI) / numSectors;
        let currentRotation = 0;
        let isSpinning = false;

        // Отрисовка колеса
        function drawWheel() {
            ctx.clearRect(0, 0, 400, 400);
            for (let i = 0; i < numSectors; i++) {
                const angle = i * arc;
                ctx.beginPath();
                ctx.fillStyle = sectors[i].color;
                ctx.moveTo(200, 200);
                ctx.arc(200, 200, 190, angle, angle + arc);
                ctx.lineTo(200, 200);
                ctx.fill();
                ctx.stroke();

                // Текст на секторах
                ctx.save();
                ctx.translate(200, 200);
                ctx.rotate(angle + arc / 2);
                ctx.textAlign = "right";
                ctx.fillStyle = "#ffffff";
                ctx.font = "bold 18px Arial";
                ctx.fillText(sectors[i].label, 170, 6);
                ctx.restore();
            }
        }

        drawWheel();

        // Вращение колеса
        spinBtn.addEventListener('click', () => {
            if (isSpinning) return;
            isSpinning = true;
            spinBtn.disabled = true;

            // Всегда подкручиваем на 10 SOL (индекс 6) для демонстрации
            const targetIndex = 6; 
            const degreesPerSector = 360 / numSectors;
            const targetDegree = 360 - (targetIndex * degreesPerSector) - (degreesPerSector / 2) - 90;
            
            const extraTurns = 5 * 360; // 5 полных оборотов
            const totalRotation = extraTurns + targetDegree;

            canvas.style.transform = `rotate(${totalRotation}deg)`;

            setTimeout(() => {
                isSpinning = false;
                winAmountText.innerText = sectors[targetIndex].label;
                modal.classList.add('active');
            }, 4000);
        });

        // Подключение кошелька (Phantom)
        connectBtn.addEventListener('click', async () => {
            if ("solana" in window) {
                try {
                    const response = await window.solana.connect();
                    alert("वॉलेट कनेक्ट हो गया: " + response.publicKey.toString());
                } catch (err) {
                    alert("कनेक्शन रद्द कर दिया गया।");
                }
            } else {
                alert("Solana वॉलेट (например, Phantom) नहीं मिला! कृपया Phantom App का उपयोग करें।");
                window.open("https://phantom.app/", "_blank");
            }
        });
    </script>
